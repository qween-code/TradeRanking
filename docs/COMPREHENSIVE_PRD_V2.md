                        else:
                            raise Exception(f"HTTP {response.status}")
            except Exception as e:
                if attempt == max_retries - 1:
                    raise
                await asyncio.sleep(backoff_factor ** attempt)

**TR-1.3: Parameter Computation Engine**
```python
# app/services/parameter_engine.py

from typing import Dict, Any, List
import networkx as nx
from app.models import ParameterDefinition, ParameterValue
from app.database import supabase

class ParameterComputationEngine:
    """
    Computes 3000+ parameters using a DAG-based dependency system.
    """

    def __init__(self):
        self.graph = nx.DiGraph()
        self._build_dependency_graph()

    def _build_dependency_graph(self):
        """Build DAG of parameter dependencies"""
        # P001 (base) → P050 (derived) → P701 (composite)
        params = ParameterDefinition.load_all()
        for param in params:
            self.graph.add_node(param.id, definition=param)
            if param.formula:
                dependencies = self._extract_dependencies(param.formula)
                for dep in dependencies:
                    self.graph.add_edge(dep, param.id)

    async def compute_for_rfq(self, rfq_id: str) -> Dict[str, Any]:
        """
        Compute all relevant parameters for an RFQ.
        Uses topological sort to compute in correct order.
        """
        # Get RFQ details
        rfq = await self._get_rfq(rfq_id)

        # Determine which parameters are relevant
        relevant_params = self._filter_relevant_params(rfq)

        # Topological sort for computation order
        compute_order = list(nx.topological_sort(
            self.graph.subgraph(relevant_params)
        ))

        # Parallel computation where possible
        results = {}
        for level in self._get_parallel_levels(compute_order):
            tasks = [
                self._compute_param(param_id, rfq, results)
                for param_id in level
            ]
            level_results = await asyncio.gather(*tasks)
            results.update(dict(zip(level, level_results)))

        return results

    async def _compute_param(
        self,
        param_id: str,
        rfq: Dict,
        computed_values: Dict
    ) -> Any:
        """Compute a single parameter"""
        definition = self.graph.nodes[param_id]['definition']

        # Check cache first
        cached = await self._get_cached_value(param_id, rfq)
        if cached and self._is_cache_valid(cached, definition.update_freq):
            return cached['value']

        # Compute based on source
        if definition.source == 'external_api':
            value = await self._fetch_from_api(definition, rfq)
        elif definition.source == 'model':
            value = await self._compute_from_model(definition, rfq, computed_values)
        elif definition.source == 'internal':
            value = await self._query_database(definition, rfq)
        else:
            value = None

        # Cache result
        await self._cache_value(param_id, rfq, value, definition)

        return value

    def _get_parallel_levels(self, ordered_params: List[str]) -> List[List[str]]:
        """
        Group parameters into levels that can be computed in parallel.
        Level N can only be computed after all levels < N.
        """
        levels = []
        processed = set()

        for param in ordered_params:
            dependencies = set(self.graph.predecessors(param))
            if dependencies.issubset(processed):
                # Can be computed in parallel with others at this level
                if not levels or not all(
                    dep in processed for dep in dependencies
                ):
                    levels.append([])
                levels[-1].append(param)
                processed.add(param)

        return levels
```

            event_type=event_type,
            severity=severity,
            latitude=event['latitude'],
            longitude=event['longitude'],
            description=event['description'],
            entities=entities,
            similar_events=similar,
            occurred_at=event['event_date'],
            ingested_at=datetime.utcnow()
        )

    async def _process_event_buffer(self):
        """Process events in batches every 60 seconds"""
        while True:
            await asyncio.sleep(60)
            if not self.event_buffer:
                continue

            # Deduplicate events
            unique_events = self._deduplicate(self.event_buffer)
            self.event_buffer = []

            # Assess impact on routes and suppliers
            for event in unique_events:
                await self._assess_impact(event)

    async def _assess_impact(self, event: GeopoliticalEvent):
        """
        Assess impact of event on:
        1. Routes (proximity to route, route dependency on affected region)
        2. Suppliers (location proximity, supplier's suppliers)
        3. Active RFQs (affected suppliers/routes)
        """
        # Find affected routes
        affected_routes = await self._find_affected_routes(event)

        # Find affected suppliers
        affected_suppliers = await self._find_affected_suppliers(event)

        # Update risk scores
        for route in affected_routes:
            impact_score = self._calculate_route_impact(event, route)
            await self._update_route_risk(route.id, impact_score, event)

        for supplier in affected_suppliers:
            impact_score = self._calculate_supplier_impact(event, supplier)
            await self._update_supplier_risk(supplier.id, impact_score, event)

        # Find affected RFQs
        affected_rfqs = await self._find_affected_rfqs(
            affected_routes, affected_suppliers
        )

        # Send alerts
        for rfq in affected_rfqs:
            await self._send_alert(rfq, event, affected_routes, affected_suppliers)

    async def _find_affected_routes(self, event: GeopoliticalEvent) -> List[Route]:
        """
        Find routes affected by event based on:
        - Geographic proximity (event within X km of route)
        - Route dependency (route passes through affected country)
        - Chokepoint proximity (event near Suez, Panama, Hormuz, etc.)
        """
        routes = await Route.get_all_active()
        affected = []

        for route in routes:
            # Check proximity
            if self._is_near_route(event, route, radius_km=500):
                affected.append(route)
                continue

            # Check if route passes through affected country
            event_country = await self._get_country(event.latitude, event.longitude)
            if event_country in route.transit_countries:
                affected.append(route)
                continue

            # Check chokepoint proximity
            if self._is_near_chokepoint(event):
                chokepoint_routes = await Route.get_by_chokepoint(
                    event.chokepoint_name
                )
                affected.extend(chokepoint_routes)

        return affected

    def _calculate_route_impact(
        self,
        event: GeopoliticalEvent,
        route: Route
    ) -> int:
        """
        Calculate impact score (0-100) based on:
        - Event severity
        - Proximity to route
        - Historical impact of similar events
        - Route importance (volume, criticality)
        """
        # Base impact from event severity
        impact = event.severity  # 0-100

        # Adjust for proximity
        distance_km = self._distance_to_route(event, route)
        proximity_factor = max(0, 1 - distance_km / 1000)  # 0 at 1000km+
        impact *= (0.5 + 0.5 * proximity_factor)

        # Adjust for historical impact
        if event.similar_events:
            avg_historical_impact = sum(
                e.actual_impact for e in event.similar_events
            ) / len(event.similar_events)
            impact = 0.7 * impact + 0.3 * avg_historical_impact

        # Adjust for route importance
        if route.is_critical:
            impact *= 1.2

        return min(100, int(impact))

from typing import List
from app.models import RFQ, GeopoliticalEvent, Route, Supplier
from app.notifications import EmailNotifier, SMSNotifier, SlackNotifier

class AlertService:
    """
    Multi-channel alert system for critical events.
    """

    def __init__(self):
        self.email = EmailNotifier()
        self.sms = SMSNotifier()
        self.slack = SlackNotifier()

    async def send_alert(
        self,
        rfq: RFQ,
        event: GeopoliticalEvent,
        affected_routes: List[Route],
        affected_suppliers: List[Supplier]
    ):
        """Send alert to RFQ owner"""
        # Check user's alert preferences
        user = await rfq.get_owner()
        prefs = user.alert_preferences

        # Check if event meets threshold
        if event.severity < prefs.min_severity:
            return

        # Aggregate related events (if multiple events in same region)
        related_events = await self._find_related_events(event, time_window_hours=6)

        # Build alert message
        message = self._build_alert_message(
            rfq, event, related_events, affected_routes, affected_suppliers
        )

        # Send via configured channels
        tasks = []
        if prefs.email_enabled:
            tasks.append(self.email.send(user.email, message))
        if prefs.sms_enabled and event.severity >= 80:
            tasks.append(self.sms.send(user.phone, message.short_summary))
        if prefs.slack_enabled:
            tasks.append(self.slack.send(user.slack_webhook, message))

        await asyncio.gather(*tasks)

        # Log alert
        await self._log_alert(rfq, event, user, message)

    def _build_alert_message(
        self,
        rfq: RFQ,
        event: GeopoliticalEvent,
        related_events: List[GeopoliticalEvent],
        affected_routes: List[Route],
        affected_suppliers: List[Supplier]
    ):
        """Build structured alert message"""
        return {
            "subject": f"🚨 High-Risk Event: {event.event_type} affecting RFQ {rfq.id}",
            "summary": f"{event.description}",
            "severity": event.severity,
            "occurred_at": event.occurred_at,
            "affected_routes": [
                {
                    "route": r.name,
                    "impact_score": r.current_risk_score,
                    "recommendation": self._get_route_recommendation(r)
                }
                for r in affected_routes
            ],
            "affected_suppliers": [
                {
                    "supplier": s.name,
                    "location": s.location,
                    "impact_score": s.current_risk_score,
                    "recommendation": self._get_supplier_recommendation(s)
                }
                for s in affected_suppliers
            ],
            "recommended_actions": self._generate_recommendations(
                rfq, event, affected_routes, affected_suppliers
            ),
            "related_events": [
                {"type": e.event_type, "severity": e.severity}
                for e in related_events
            ]
        }

    def _generate_recommendations(
        self,
        rfq: RFQ,
        event: GeopoliticalEvent,
        affected_routes: List[Route],
        affected_suppliers: List[Supplier]
    ) -> List[str]:
        """Generate actionable recommendations"""
        recommendations = []

        # Route alternatives
        if affected_routes:
            for route in affected_routes:
                alternatives = route.get_alternatives()
                if alternatives:
                    best_alt = min(alternatives, key=lambda r: r.risk_score)
                    recommendations.append(
                        f"Consider switching from {route.name} to {best_alt.name} "
                        f"(+{best_alt.cost_delta_percent:.1f}% cost, "
                        f"+{best_alt.time_delta_days} days)"
                    )

        # Supplier alternatives
        if affected_suppliers:
            for supplier in affected_suppliers:
                alternatives = await self._find_alternative_suppliers(
                    rfq, exclude=[supplier.id]
                )
                if alternatives:
                    best_alt = alternatives[0]
                    recommendations.append(
                        f"Consider switching from {supplier.name} to {best_alt.name} "
                        f"(Risk score: {best_alt.risk_score}/100)"
                    )

        # Timing recommendations
        if event.severity >= 70:
            recommendations.append(
                f"Consider delaying RFQ deadline by {self._suggest_delay_days(event)} days "
                f"to allow situation to stabilize"
            )

        return recommendations
